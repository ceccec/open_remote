def update_performance_ratio!
  return unless total_capacity.to_f.positive?

  ratio = (total_power_output.to_f / total_capacity.to_f) * 100.0
  self.attributes_data ||= {}
  attributes_data["performanceRatio"] = ratio
  save!
end
